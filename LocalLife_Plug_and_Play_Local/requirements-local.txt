# YOLOE's text-prompt mode (curated bag/box vocabulary, not the broad
# prompt-free fallback) needs `import clip` at load time. Ultralytics only
# ships that as `git+https://github.com/ultralytics/CLIP.git`, which needs a
# `git` executable on PATH and a working github.com connection -- neither is
# guaranteed on a plain Windows laptop, and when it fails Ultralytics silently
# swaps in the prompt-free fallback (see CHANGES_local-ai-19.2.0.md, Bug 1).
# `openai-clip` is a plain-PyPI mirror of the same `clip` package (identical
# `clip.load()` / `clip.tokenize()` API, confirmed by inspecting both
# side-by-side) -- installing it here means `import clip` just succeeds on
# pip's normal first attempt and Ultralytics' git-based fallback path never
# runs at all.
openai-clip>=1.0,<2
ftfy>=6.1
regex>=2023.0
tqdm>=4.64

# openai-clip's own clip.py still does `from pkg_resources import packaging`
# at import time. Newer setuptools (81+) removed pkg_resources entirely, so
# without this pin `import clip` throws ModuleNotFoundError on a fresh venv
# that happens to get a current setuptools -- confirmed directly in this
# sandbox: setuptools 84 breaks the import, setuptools<81 fixes it.
setuptools<81
